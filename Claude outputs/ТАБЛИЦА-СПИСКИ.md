# Починка выпадающих списков в реестре

## Что сломалось

Списки в таблице собраны не из перечисления значений, а из **именованных
диапазонов** — коротких имён вроде `ВИДЫ_РАБОТ`, за которыми стоит кусок листа
«Справочники». При переносе из Excel в Google уцелело ровно одно имя из
четырнадцати — `ЗАКАЗЧИКИ_СПИСОК`. Остальные тринадцать не существуют, поэтому
проверка данных ссылается в пустоту, и список не раскрывается.

Формулы при этом целы: ни одной ячейки с ошибкой во всей книге, `TEXTJOIN`
Google понимает. Ломались только имена — их Google при импорте теряет чаще всего.

Что не работает сейчас:

| Лист | Столбец | Список |
|---|---|---|
| Объекты | F — Тип здания | `ТИП_ЗДАНИЯ` |
| Объекты | H — Ед. изм. | `ЕДИНИЦЫ` |
| Договора | C — Вид работы | `ВИДЫ_РАБОТ` |
| Договора | B — ОБЪЕКТ | `ОБЪЕКТЫ_СПИСОК` |
| Договора | D — Исполнитель (юр. лицо) | `ЮР_ЛИЦА` |
| Договора | E — Статус договора | `СТАТУС_ЭТАПА` |
| Договора | J:M — Ответственный, соисполнители | `ИСПОЛНИТЕЛИ` |
| Договора | N, Q, T — условия оплаты | `ЭТАПЫ_ОПЛАТЫ` |
| Сдача | B — ДОГОВОР | `ДОГОВОРА_СПИСОК` |
| Сдача | F — Тип | `ТИП_СДАЧИ` |
| Сдача | I — Статус сдачи | `СТАТУС_СДАЧИ` |
| Сдача | J — Ответственный | `ИСПОЛНИТЕЛИ` |
| Сотрудники | A — Отдел | `ОТДЕЛЫ` |
| Справочники | B6:B55 — Стадия | `СТАДИЯ_ОБЪЕКТА` |
| Карточки договоров | B4 — выбор объекта | `ОБЪЕКТЫ_СПИСОК` |

## Как починить

1. В таблице: **Расширения → Apps Script**.
2. Вставить код ниже отдельным файлом (или поверх пустого `Код.gs`), сохранить.
3. Сверху выбрать функцию `починитьСписки` и нажать **Выполнить**.
4. Один раз Google спросит разрешение на доступ к таблице — разрешить.

Скрипт создаёт недостающие имена и заново вешает проверку данных на нужные
столбцы. Данные он не трогает: пишет только правила, ни одной ячейки со
значением не меняет. Запускать можно повторно — он идемпотентен.

```javascript
/**
 * Восстанавливает именованные диапазоны и выпадающие списки реестра.
 * Значения в ячейках не меняются — только имена и правила проверки данных.
 */

var ИМЕНА = {
  // Справочники — короткие списки
  'СТАТУС_ЭТАПА':    'Справочники!$A$6:$A$46',
  'ВИДЫ_РАБОТ':      'Справочники!$A$60:$A$81',
  'ЭТАПЫ_ОПЛАТЫ':    'Справочники!$A$94:$A$103',
  'ЕДИНИЦЫ':         'Справочники!$B$94:$B$100',
  'ЮР_ЛИЦА':         'Справочники!$C$94:$C$96',
  'ТИП_ЗДАНИЯ':      'Справочники!$D$94:$D$108',
  'ТИП_СДАЧИ':       'Справочники!$E$94:$E$97',
  'СТАТУС_СДАЧИ':    'Справочники!$F$94:$F$101',
  'СТАДИЯ_ОБЪЕКТА':  'Справочники!$G$94:$G$99',
  'ОТДЕЛЫ':          'Справочники!$H$94:$H$101',

  // Живые списки — тянутся из рабочих листов
  'ЗАКАЗЧИКИ_СПИСОК': 'Заказчики!$B$4:$B$628',
  'ОБЪЕКТЫ_СПИСОК':   'Объекты!$B$5:$B$54',
  'ДОГОВОРА_СПИСОК':  'Договора!$AM$5:$AM$104',
  'ИСПОЛНИТЕЛИ':      'Сотрудники!$B$5:$B$104'
};

/** Куда какой список вешается. */
var СПИСКИ = [
  { лист: 'Объекты',            диапазон: 'D5:D54',                имя: 'ЗАКАЗЧИКИ_СПИСОК' },
  { лист: 'Объекты',            диапазон: 'F5:F54',                имя: 'ТИП_ЗДАНИЯ' },
  { лист: 'Объекты',            диапазон: 'H5:H54',                имя: 'ЕДИНИЦЫ' },

  { лист: 'Договора',           диапазон: 'B5:B104',               имя: 'ОБЪЕКТЫ_СПИСОК' },
  { лист: 'Договора',           диапазон: 'C5:C104',               имя: 'ВИДЫ_РАБОТ' },
  { лист: 'Договора',           диапазон: 'D5:D104',               имя: 'ЮР_ЛИЦА' },
  { лист: 'Договора',           диапазон: 'E5:E104',               имя: 'СТАТУС_ЭТАПА' },
  { лист: 'Договора',           диапазон: 'J5:M104',               имя: 'ИСПОЛНИТЕЛИ' },
  { лист: 'Договора',           диапазон: 'N5:N104',               имя: 'ЭТАПЫ_ОПЛАТЫ' },
  { лист: 'Договора',           диапазон: 'Q5:Q104',               имя: 'ЭТАПЫ_ОПЛАТЫ' },
  { лист: 'Договора',           диапазон: 'T5:T104',               имя: 'ЭТАПЫ_ОПЛАТЫ' },

  { лист: 'Сдача',              диапазон: 'B5:B124',               имя: 'ДОГОВОРА_СПИСОК' },
  { лист: 'Сдача',              диапазон: 'F5:F124',               имя: 'ТИП_СДАЧИ' },
  { лист: 'Сдача',              диапазон: 'I5:I124',               имя: 'СТАТУС_СДАЧИ' },
  { лист: 'Сдача',              диапазон: 'J5:J124',               имя: 'ИСПОЛНИТЕЛИ' },

  { лист: 'Карточки договоров', диапазон: 'B4',                    имя: 'ОБЪЕКТЫ_СПИСОК' },
  { лист: 'Сотрудники',         диапазон: 'A5:A104',               имя: 'ОТДЕЛЫ' },
  { лист: 'Справочники',        диапазон: 'B6:B55',                имя: 'СТАДИЯ_ОБЪЕКТА' },
  { лист: 'Справочники',        диапазон: 'B60:B89',               имя: 'ОТДЕЛЫ' }
];

function починитьСписки() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var отчёт = [];

  // 1. Имена
  var было = {};
  ss.getNamedRanges().forEach(function (nr) { было[nr.getName()] = true; });

  Object.keys(ИМЕНА).forEach(function (имя) {
    var адрес = ИМЕНА[имя];
    var диапазон;
    try {
      диапазон = ss.getRange(адрес);
    } catch (e) {
      отчёт.push('пропущено ' + имя + ' — нет такого диапазона (' + адрес + ')');
      return;
    }
    // setNamedRange перезаписывает имя, если оно уже есть
    ss.setNamedRange(имя, диапазон);
    отчёт.push((было[имя] ? 'обновлено ' : 'создано  ') + имя + '  ->  ' + адрес);
  });

  // 2. Проверка данных
  СПИСКИ.forEach(function (с) {
    var лист = ss.getSheetByName(с.лист);
    if (!лист) {
      отчёт.push('пропущен лист ' + с.лист);
      return;
    }
    var источник;
    try {
      источник = ss.getRange(ИМЕНА[с.имя]);
    } catch (e) {
      отчёт.push('нет источника для ' + с.имя);
      return;
    }
    var правило = SpreadsheetApp.newDataValidation()
      .requireValueInRange(источник, true)   // true — показывать выпадающий список
      .setAllowInvalid(false)
      .setHelpText('Значение выбирается из списка «' + с.имя + '» на листе «Справочники»')
      .build();
    лист.getRange(с.диапазон).setDataValidation(правило);
    отчёт.push('список  ' + с.лист + '!' + с.диапазон + '  <-  ' + с.имя);
  });

  Logger.log(отчёт.join('\n'));
  SpreadsheetApp.getActive().toast(
    'Готово: имён ' + Object.keys(ИМЕНА).length + ', списков ' + СПИСКИ.length,
    'Списки восстановлены', 8
  );
}

/** Показать, что сейчас есть в таблице — для проверки до и после. */
function показатьИмена() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var строки = ss.getNamedRanges().map(function (nr) {
    return nr.getName() + '  ->  ' + nr.getRange().getA1Notation();
  });
  Logger.log(строки.length ? строки.join('\n') : 'именованных диапазонов нет');
}
```

## Проверка

После запуска открой `Объекты`, встань на любую ячейку в столбце **Тип здания** —
должна появиться стрелка со списком из пятнадцати типов. В `Договорах` столбец
**Статус договора** должен дать сорок один статус, от «Потенциальный» до
«Расторжение договора».

Если какой-то список окажется короче, чем ожидаешь, — значит в «Справочниках»
добавлены строки ниже границы диапазона. Поправь адрес в блоке `ИМЕНА` и запусти
функцию ещё раз, данные от этого не пострадают.

## На будущее

Именованные диапазоны Google теряет почти при каждом импорте из Excel. Если
будешь ещё раз переносить книгу — сначала запусти `показатьИмена` на исходнике,
сохрани список, а после переноса прогони `починитьСписки`.
